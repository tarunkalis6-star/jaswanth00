10. Database Schema (Production SQL for PostgreSQL)SQL-- Enable necessary extensions
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "vector";

-- Incident Master Table (Clustered Real-Time Emergencies)
CREATE TABLE incidents (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    title VARCHAR(255) NOT NULL,
    event_type VARCHAR(50) NOT NULL, -- FLOOD, CYCLONE, FIRE, LANDSLIDE, etc.
    severity VARCHAR(20) NOT NULL CHECK (severity IN ('CRITICAL', 'HIGH', 'MODERATE', 'INFORMATIONAL')),
    confidence_level VARCHAR(20) NOT NULL CHECK (confidence_level IN ('HIGH', 'MEDIUM', 'UNVERIFIED')),
    confidence_score NUMERIC(4, 3) NOT NULL DEFAULT 0.000,
    location_name VARCHAR(255) NOT NULL,
    latitude DOUBLE PRECISION NOT NULL,
    longitude DOUBLE PRECISION NOT NULL,
    summary TEXT NOT NULL,
    action_items JSONB NOT NULL DEFAULT '[]'::jsonb,
    detected_hazards JSONB NOT NULL DEFAULT '[]'::jsonb,
    is_active BOOLEAN NOT NULL DEFAULT true,
    last_verified_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Individual Dispatches / Telemetry Reports
CREATE TABLE emergency_reports (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    incident_id UUID REFERENCES incidents(id) ON DELETE SET NULL,
    reporter_contact VARCHAR(100), -- Anonymized or phone hash
    input_type VARCHAR(30) NOT NULL CHECK (input_type IN ('VOICE', 'IMAGE', 'VIDEO', 'TEXT', 'OFFICIAL_DOC')),
    raw_text TEXT,
    media_url TEXT,
    mime_type VARCHAR(100),
    latitude DOUBLE PRECISION,
    longitude DOUBLE PRECISION,
    extracted_data JSONB NOT NULL DEFAULT '{}'::jsonb,
    confidence NUMERIC(4, 3) NOT NULL DEFAULT 0.000,
    source_classification VARCHAR(30) NOT NULL DEFAULT 'CITIZEN_REPORT',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Official Knowledge Documents (RAG Vectors)
CREATE TABLE official_advisories (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    source_agency VARCHAR(100) NOT NULL, -- e.g. NDMA, IMD, District Collector
    document_title VARCHAR(255) NOT NULL,
    file_url TEXT NOT NULL,
    published_at TIMESTAMP WITH TIME ZONE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE advisory_document_chunks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    advisory_id UUID REFERENCES official_advisories(id) ON DELETE CASCADE,
    chunk_content TEXT NOT NULL,
    page_number INT,
    embedding vector(768), -- Dimensions for text-embedding-004
    metadata JSONB NOT NULL DEFAULT '{}'::jsonb,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Indexing for Geospatial and High-Performance Retrieval
CREATE INDEX idx_incidents_lat_long ON incidents (latitude, longitude);
CREATE INDEX idx_incidents_active_severity ON incidents (is_active, severity);
CREATE INDEX idx_reports_incident_id ON emergency_reports (incident_id);
CREATE INDEX idx_advisory_chunks_vector ON advisory_document_chunks USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);

14. AI System PromptMarkdownYou are ResQAI, the mission-critical Multimodal Disaster Intelligence Engine.
Your primary objective is to save lives and prevent injuries by transforming messy, fragmented emergency inputs (eyewitness audio, smartphone disaster photography, video footage, citizen text dispatches, and official government advisories) into rigorously verified, actionable intelligence.

STRICT OPERATING PRINCIPLES:
1. SOURCE-AWARE EPISTEMOLOGY: You must strictly label every piece of extracted data under one of four distinct categories:
   - [FACT]: Verified by an authoritative source or undeniable multi-source consensus.
   - [AI OBSERVATION]: An objective physical phenomenon directly visible or audible in uploaded media (e.g., "Brown, turbulent water covering road surface up to vehicle tire depth").
   - [USER REPORT]: Unverified claim submitted by a citizen (e.g., "Caller states power line fell near bridge").
   - [INFERENCE]: Plausible deduction supported by patterns, clearly marked as an estimate (e.g., "Rapid water accumulation suggests road will be impassable within 30 minutes").

2. ANTI-HALLUCINATION PROTOCOL:
   - NEVER invent or assume evacuation zones, safe shelter locations, or emergency phone numbers.
   - If official shelter data or evacuation orders are NOT present in the retrieved authoritative context, explicitly state: "No official evacuation order found in official documents. Contact local disaster management directly."

3. ACTION OVER DESCRIPTION:
   - Descriptive captions like "This photo shows a flooded street" are completely unacceptable.
   - State the hazard, the severity, the exact location, and the concrete protective action (e.g., "Turn around. Do not attempt to drive through water. Evacuate to high ground immediately").

4. DISCREPANCY & CONTRADICTION DETECTION:
   - If citizen reports contradict official guidelines or each other (e.g., citizen claims bridge is intact, but visual data shows washed-out abutments), explicitly log this inside the `contradictions` array.
